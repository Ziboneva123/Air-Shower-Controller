INTELLIGENT AIR SHOWER CONTROLLER V4.7

Arduino Mega 2560 • Sensor 12V • Relay 12V Active HIGH

1. Fungsi Sistem

Controller mengatur interlock dua pintu, deteksi orang, blower, lighting, buzzer, E-Stop dan alarm.

Sensor 12V → Relay input 12V → kontak NO/COM → Arduino D30-D35 → Arduino D22-D26 → Relay output 12V → actuator
JANGAN MASUKKAN 12V KE PIN ARDUINO. D30-D35 hanya menerima level 0/5V dari kontak relay/interface.

2. Tabel I/O

PinFungsiLogika
D30Sensor 1HIGH = person detected
D31Sensor 2HIGH = person detected
D32Limit outerHIGH = door closed
D33Limit innerHIGH = door closed
D34E-StopHIGH = active
D35ResetHIGH = pressed
D22Outer lockHIGH = LOCK
D23Inner lockHIGH = LOCK
D24BlowerHIGH = ON
D25LightingLOW = ON
D26BuzzerLOW = ON

3. Sequence

  1. IDLE: kedua pintu dapat dibuka, blower OFF, lighting ON.
  2. Outer dibuka lebih dulu → inner LOCK.
  3. Inner dibuka lebih dulu → outer LOCK, tanpa shower.
  4. Sensor 1 mendeteksi orang → latch entry.
  5. Outer tertutup → kedua pintu LOCK, menunggu Sensor 2.
  6. Sensor 2 aktif → person latch → blower ON 20 detik.
  7. 20 detik selesai → blower OFF, outer tetap LOCK, inner UNLOCK.
  8. Inner dibuka lalu ditutup → siklus selesai → IDLE.

4. Timer & Alarm

ParameterNilai
Shower20 s
Entry timeout15 s
Door open timeout30 s
After shower timeout30 s
Reset hold1 s

Alarm: E-Stop aktif, pintu terbuka terlalu lama, Sensor 2 tidak aktif setelah entry, inner tidak dibuka setelah shower, atau inner tidak tertutup setelah exit.

Saat alarm: blower OFF, kedua lock LOCK, buzzer ON, lighting OFF.

5. Wiring Interface

SisiHubungan
Sensor 12V+12V / 0V / OUT → input relay 12V
Relay inputKontak NO/COM → Arduino D30-D35 sebagai dry contact 0/5V
Arduino outputD22-D26 → driver/relay input 12V Active HIGH
Relay outputCOM/NO/NC → beban sesuai tegangan dan rating aktual
Gunakan driver transistor/MOSFET bila input relay output membutuhkan 12V dan tidak dapat dipicu langsung oleh 5V Arduino. Tegangan lock/blower/lampu harus mengikuti nameplate aktual.

6. Source Code Arduino Mega V4.7

Buka/tutup coding
// INTELLIGENT AIR SHOWER CONTROLLER - ARDUINO MEGA 2560
// SOFTWARE V4.7 FINAL FIXED
//
// INPUT: D30 Sensor1, D31 Sensor2, D32 OuterLimit, D33 InnerLimit,
//        D34 E-Stop, D35 Reset
// OUTPUT: D22 OuterLock, D23 InnerLock, D24 Blower,
//         D25 Lighting, D26 Buzzer
// INPUTS are 0/5V from relay dry contacts. NEVER connect 12V directly.

enum State { IDLE, OUTER_DOOR_ACTIVE, INNER_DOOR_ACTIVE,
             ENTRY_WAIT_PERSON, SHOWER, AFTER_SHOWER, EXIT_SHOWER, ALARM };

const byte OUTER_LOCK=22, INNER_LOCK=23, BLOWER=24, LIGHT=25, BUZZER=26;
const byte S1=30, S2=31, OUTER_LIMIT=32, INNER_LIMIT=33, ESTOP=34, RESET=35;

const unsigned long SHOWER_TIME=20000UL;
const unsigned long DOOR_TIMEOUT=30000UL;
const unsigned long ENTRY_TIMEOUT=15000UL;
const unsigned long AFTER_SHOWER_TIMEOUT=30000UL;
const unsigned long RESET_HOLD=1000UL;

State state=IDLE;
bool sensor1,sensor2,outerClosed,innerClosed,eStop;
bool entryDetected=false, personDetected=false;
unsigned long stateStart=0, showerStart=0, afterShowerStart=0, resetStart=0;

void lockOuter(){digitalWrite(OUTER_LOCK,HIGH);}
void unlockOuter(){digitalWrite(OUTER_LOCK,LOW);}
void lockInner(){digitalWrite(INNER_LOCK,HIGH);}
void unlockInner(){digitalWrite(INNER_LOCK,LOW);}
void blowerON(){digitalWrite(BLOWER,HIGH);}
void blowerOFF(){digitalWrite(BLOWER,LOW);}
void lightON(){digitalWrite(LIGHT,LOW);}
void lightOFF(){digitalWrite(LIGHT,HIGH);}
void buzzerON(){digitalWrite(BUZZER,LOW);}
void buzzerOFF(){digitalWrite(BUZZER,HIGH);}

void alarmNow(){
  state=ALARM; stateStart=millis();
  blowerOFF(); lockOuter(); lockInner(); lightOFF(); buzzerON();
}

void readInputs(){
  sensor1=digitalRead(S1)==HIGH;
  sensor2=digitalRead(S2)==HIGH;
  outerClosed=digitalRead(OUTER_LIMIT)==HIGH;
  innerClosed=digitalRead(INNER_LIMIT)==HIGH;
  eStop=digitalRead(ESTOP)==HIGH;
}

bool resetCycle(){
  if(personDetected){ alarmNow(); return false; }
  entryDetected=false; personDetected=false;
  blowerOFF(); buzzerOFF(); unlockOuter(); unlockInner(); lightON();
  resetStart=0; state=IDLE; stateStart=millis(); return true;
}

void idleState(){
  blowerOFF(); buzzerOFF(); lightON();
  if(!outerClosed){ lockInner(); state=OUTER_DOOR_ACTIVE; stateStart=millis(); return; }
  if(!innerClosed){ lockOuter(); state=INNER_DOOR_ACTIVE; stateStart=millis(); return; }
  unlockOuter(); unlockInner();
  if(sensor1){entryDetected=true; lockInner(); state=ENTRY_WAIT_PERSON; stateStart=millis();}
}

void outerState(){
  lockInner(); unlockOuter();
  if(millis()-stateStart>=DOOR_TIMEOUT){alarmNow();return;}
  if(outerClosed){
    if(sensor1){entryDetected=true;lockInner();state=ENTRY_WAIT_PERSON;stateStart=millis();}
    else state=IDLE;
  }
}

void innerState(){
  lockOuter(); unlockInner();
  if(millis()-stateStart>=DOOR_TIMEOUT){alarmNow();return;}
  if(innerClosed) resetCycle();
}

void entryState(){
  lockInner(); unlockOuter(); blowerOFF(); lightON();
  if(!entryDetected){state=IDLE;return;}
  if(sensor2){personDetected=true;showerStart=millis();blowerON();lockOuter();lockInner();state=SHOWER;stateStart=millis();return;}
  if(millis()-stateStart>=ENTRY_TIMEOUT) alarmNow();
}

void showerState(){
  lockOuter(); lockInner(); blowerON(); lightON();
  if(millis()-showerStart>=SHOWER_TIME){
    blowerOFF(); lockOuter(); unlockInner();
    afterShowerStart=millis(); state=AFTER_SHOWER; stateStart=millis();
  }
}

void afterShowerState(){
  blowerOFF(); lockOuter(); unlockInner(); lightON();
  if(!innerClosed){state=EXIT_SHOWER;stateStart=millis();return;}
  if(millis()-afterShowerStart>=AFTER_SHOWER_TIMEOUT) alarmNow();
}

void exitState(){
  blowerOFF(); lockOuter(); unlockInner(); lightON();
  if(millis()-stateStart>=DOOR_TIMEOUT){alarmNow();return;}
  if(innerClosed){personDetected=false;resetCycle();}
}

void alarmState(){
  blowerOFF(); lockOuter(); lockInner(); lightOFF(); buzzerON();
  if(!eStop && outerClosed && innerClosed && !sensor2){
    if(digitalRead(RESET)==HIGH){
      if(resetStart==0) resetStart=millis();
      if(millis()-resetStart>=RESET_HOLD){
        entryDetected=false;personDetected=false;buzzerOFF();
        unlockOuter();unlockInner();lightON();
        resetStart=0;state=IDLE;stateStart=millis();
      }
    } else resetStart=0;
  } else resetStart=0;
}

void setup(){
  pinMode(OUTER_LOCK,OUTPUT); pinMode(INNER_LOCK,OUTPUT);
  pinMode(BLOWER,OUTPUT); pinMode(LIGHT,OUTPUT); pinMode(BUZZER,OUTPUT);
  pinMode(S1,INPUT);pinMode(S2,INPUT);pinMode(OUTER_LIMIT,INPUT);
  pinMode(INNER_LIMIT,INPUT);pinMode(ESTOP,INPUT);pinMode(RESET,INPUT);
  unlockOuter();unlockInner();blowerOFF();lightON();buzzerOFF();
  Serial.begin(9600); state=IDLE; stateStart=millis();
}

void loop(){
  readInputs();
  if(eStop && state!=ALARM) alarmNow();
  if(state==ENTRY_WAIT_PERSON && sensor2){
    personDetected=true; showerStart=millis(); blowerON(); lockOuter(); lockInner();
    state=SHOWER; stateStart=millis();
  }
  switch(state){
    case IDLE: idleState(); break;
    case OUTER_DOOR_ACTIVE: outerState(); break;
    case INNER_DOOR_ACTIVE: innerState(); break;
    case ENTRY_WAIT_PERSON: entryState(); break;
    case SHOWER: showerState(); break;
    case AFTER_SHOWER: afterShowerState(); break;
    case EXIT_SHOWER: exitState(); break;
    case ALARM: alarmState(); break;
  }
}

7. Catatan Keselamatan

  • 12V sensor dan coil relay harus mempunyai supply/fuse yang sesuai.
  • Jangan hubungkan 12V langsung ke Arduino Mega.
  • Input Arduino harus mempunyai level LOW/HIGH yang pasti, tidak floating.
  • Beban AC harus melalui relay/contactor yang ratingnya sesuai.
  • E-Stop sebaiknya juga memutus actuator secara hardware/safety circuit.
  • Lakukan FAT setelah wiring: semua sensor, interlock, blower 20s, timeout, E-Stop dan reset.
